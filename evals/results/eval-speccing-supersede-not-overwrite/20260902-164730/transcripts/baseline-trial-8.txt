import io, sys

def edit(path, replacements, append=None):
    with io.open(path, encoding="utf-8") as f:
        s = f.read()
    for old, new in replacements:
        if s.count(old) != 1:
            sys.exit(f"{path}: expected exactly 1 match for {old!r}, found {s.count(old)}")
        s = s.replace(old, new)
    if append:
        if not s.endswith("\n"):
            s += "\n"
        s += append
    with io.open(path, "w", encoding="utf-8", newline="\n") as f:
        f.write(s)
