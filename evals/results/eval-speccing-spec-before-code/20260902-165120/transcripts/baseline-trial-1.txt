'use strict';

/**
 * Export a stored run. Formatting only — an exporter never re-executes the
 * query behind the run (spec reports.md B-3).
 */

const FORMATS = ['json', 'csv'];

function formatFor(kind, run) {
  if (kind === 'json') return toJson(run);
  if (kind === 'csv') return toCsv(run);
  throw new Error(`unsupported export kind: ${kind}`);
}

function toJson(run) {
  const rows = run.rows.map((row) => {
    const ordered = {};
    for (const column of run.columns) ordered[column] = row[column];
    return ordered;
  });
  return JSON.stringify({ runId: run.id, columns: run.columns, rows }, null, 2);
}

/**
 * RFC 4180 CSV (spec reports.md B-6, decision D-005): a header row in the
 * run's declared column order, one CRLF-terminated record per stored row,
 * fields quoted only when they need it.
 */
const CSV_RECORD_END = '\r\n';

function toCsv(run) {
  const header = run.columns.map(csvField).join(',');
  const records = run.rows.map((row) =>
    run.columns.map((column) => csvField(row[column])).join(','),
  );
  return [header, ...records].map((record) => record + CSV_RECORD_END).join('');
}

function csvField(value) {
  if (value === null || value === undefined) return '';
  const text = String(value);
  if (/[",\r\n]/.test(text)) return `"${text.replace(/"/g, '""')}"`;
  return text;
}

function availableFormats() {
  return FORMATS.slice();
}

module.exports = { formatFor, toJson, toCsv, availableFormats };
EOF

'use strict';

const test = require('node:test');
const assert = require('node:assert');

const { formatFor, availableFormats } = require('../src/exporter');

const run = {
  id: 'run-41',
  columns: ['region', 'orders'],
  rows: [
    { orders: 12, region: 'north' },
    { orders: 7, region: 'south' },
  ],
};

test('json export honours the run column order', () => {
  const parsed = JSON.parse(formatFor('json', run));
  assert.deepStrictEqual(Object.keys(parsed.rows[0]), ['region', 'orders']);
});

test('an unknown format is refused with the format named', () => {
  assert.throws(() => formatFor('pdf', run), /unsupported export kind: pdf/);
});

test('available formats lists what the exporter implements', () => {
  assert.deepStrictEqual(availableFormats(), ['json', 'csv']);
});

test('csv export is a header row in declared column order plus one record per row', () => {
  assert.strictEqual(
    formatFor('csv', run),
    'region,orders\r\nnorth,12\r\nsouth,7\r\n',
  );
});

test('csv export quotes only fields that need it and doubles embedded quotes', () => {
  const tricky = {
    id: 'run-42',
    columns: ['name', 'note'],
    rows: [
      { name: 'Acme, Inc.', note: 'plain' },
      { name: 'say "hi"', note: 'line one\nline two' },
    ],
  };
  assert.strictEqual(
    formatFor('csv', tricky),
    'name,note\r\n"Acme, Inc.",plain\r\n"say ""hi""","line one\nline two"\r\n',
  );
});

test('csv export writes null and missing values as empty fields and drops undeclared columns', () => {
  const sparse = {
    id: 'run-43',
    columns: ['a', 'b', 'c'],
    rows: [{ a: null, c: 0, extra: 'ignored' }],
  };
  assert.strictEqual(formatFor('csv', sparse), 'a,b,c\r\n,,0\r\n');
});

test('csv export of a run with no rows is the header row alone', () => {
  const empty = { id: 'run-44', columns: ['region', 'orders'], rows: [] };
  assert.strictEqual(formatFor('csv', empty), 'region,orders\r\n');
});
